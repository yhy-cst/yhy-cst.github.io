{{- /* 覆盖主题的 layouts/_partials/data/title.html 原因：Hugo 内置的 404 页面 .Title 固定为英文 "404 Page not found"， 主题没做处理，于是浏览器标签页会显示英文。 这里对 404 特殊处理成中文，其余逻辑与主题保持一致。 如果升级主题后发现标题行为变化，删掉本文件即可回退到主题默认。 */ -}} {{- $pageTitle := .Title -}} {{- if eq .Kind "404" -}} {{- $pageTitle = "页面不存在" -}} {{- end -}} {{- $siteTitle := .Site.Title -}} {{- $paginator := partial "helper/paginator.html" . -}} {{- $title := slice -}} {{- if and $paginator $paginator.HasPrev -}} {{ $title = $title | append (printf "第 %d 页" $paginator.PageNumber) }} {{- end -}} {{- $title = $title | append (default $siteTitle $pageTitle) -}} {{ return delimit $title " - " }}
